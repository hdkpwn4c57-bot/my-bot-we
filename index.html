import os import logging from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update from telegram.ext import Application, CallbackQueryHandler, CommandHandler, ContextTypes logging.basicConfig( format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO ) # ใส่ Token ของคุณที่นี่ (หรือตั้งค่าผ่าน Environment Variable ชื่อ TELEGRAM_BOT_TOKEN) TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN", "YOUR_TELEGRAM_BOT_TOKEN") user_sessions = {} def init_session(start_bal=255.00, base_bet=1.00): return { "history": [], # ประวัติไพ่: P, B, T "wl_history": [], # ประวัติผลไม้: W, L, - "start_bal": float(start_bal), "now_bal": float(start_bal), "base": float(base_bet), "step": 1, "units": 0, "wait_cooldown": 0 # พักไม้หลังชนะ } def analyze_next_move(s): """วิเคราะห์ไม้ต่อไปจากประวัติไพ่""" pb_only = [x for x in s["history"] if x in ["P", "B"]] # รอเก็บข้อมูลครบ 6 ตาแรก หรือกำลังอยู่ในช่วงพักไม้ (WAIT) if len(pb_only) < 6 or s["wait_cooldown"] > 0: return "WAIT", False # วิเคราะห์สูตร CHOP (แทงสลับจากตาสุดท้าย) last_side = pb_only[-1] next_side = "BANKER" if last_side == "P" else "PLAYER" return next_side, True def process_move(s, action): """ประมวลผลเมื่อผู้ใช้กดปุ่ม P, B หรือ T""" signal_side, is_active = analyze_next_move(s) # กรณีออกเสมอ (T) -> ไม่ได้ไม่เสีย บันทึก '-' if action == "T": s["wl_history"].append("-") s["history"].append("T") return # กรณีออก P หรือ B if is_active: current_stake = s["base"] * (2 ** (s["step"] - 1)) predicted = "P" if signal_side == "PLAYER" else "B" if action == predicted: # ชนะ (W): บวก 1 Unit, คำนวณกำไร (Banker หัก 5%), รีเซ็ต Step และเข้าโหมด WAIT 1 ตา payout = current_stake * 0.95 if predicted == "B" else current_stake s["now_bal"] += payout s["units"] += 1 s["step"] = 1 s["wl_history"].append("W") s["wait_cooldown"] = 1 else: # แพ้ (L): ลบ 1 Unit, หักทุน, ขยับ Step ทบไม้ถัดไป s["now_bal"] -= current_stake s["units"] -= 1 s["step"] += 1 s["wl_history"].append("L") else: # ช่วง WAIT (ไม่มีไม้) s["wl_history"].append("-") if s["wait_cooldown"] > 0: s["wait_cooldown"] -= 1 s["history"].append(action) def get_keyboard(): return InlineKeyboardMarkup([ [ InlineKeyboardButton("🔵 P (เพลเยอร์)", callback_data="P"), InlineKeyboardButton("🔴 B (แบงค์เกอร์)", callback_data="B"), InlineKeyboardButton("🟢 T (เสมอ)", callback_data="T"), ], [ InlineKeyboardButton("↩️ ย้อนกลับ (Undo)", callback_data="UNDO"), InlineKeyboardButton("🔄 ล้างกระดาน (Reset)", callback_data="RESET"), ] ]) def render_message(chat_id): s = user_sessions.get(chat_id, init_session()) p_count = s["history"].count("P") b_count = s["history"].count("B") t_count = s["history"].count("T") total_hand = len(s["history"]) # คำนวณเปอร์เซ็นต์การออกไพ่ p_pct = (p_count / total_hand * 100) if total_hand > 0 else 0 b_pct = (b_count / total_hand * 100) if total_hand > 0 else 0 # สถิติ W / L wins = s["wl_history"].count("W") losses = s["wl_history"].count("L") total_bets = wins + losses win_rate = int(round((wins / total_bets) * 100)) if total_bets > 0 else 0 # วิเคราะห์ไม้ต่อไป next_side, is_active = analyze_next_move(s) if not is_active: signal_display = "⚪️ WAIT (ไม่มีไม้ - รอจังหวะ)" stake_display = "0.00" else: icon = "🔵" if next_side == "PLAYER" else "🔴" signal_display = f"{icon} **{next_side}**" stake_val = s["base"] * (2 ** (s["step"] - 1)) stake_display = f"{stake_val:.2f} (ไม้ที่ {s['step']})" profit = s["now_bal"] - s["start_bal"] recent_road = " ".join(s["history"][-12:]) if s["history"] else "ยังไม่มีข้อมูล" recent_wl = " ".join(s["wl_history"][-10:]) if s["wl_history"] else "-" return ( f"🎰 **ระบบวิเคราะห์ไม้ & เดินเงิน (RIGEL MM)**\n" f"━━━━━━━━━━━━━━━━━━━━\n" f"🔵 P: `{p_count}` ({p_pct:.0f}%) | 🔴 B: `{b_count}` ({b_pct:.0f}%) | 🟢 T: `{t_count}` | ตาที่: `{total_hand}`\n" f"🎯 ชนะ: `{wins}` | แพ้: `{losses}` | Win Rate: `{win_rate}%`\n" f"━━━━━━━━━━━━━━━━━━━━\n" f"📡 **วิเคราะห์ไม้ต่อไป:** {signal_display}\n" f"💵 **ยอดเดินเงิน (STAKE):** `{stake_display}`\n" f"━━━━━━━━━━━━━━━━━━━━\n" f"💰 ทุน: `{s['start_bal']:.2f}` ➔ ปัจจุบัน: `{s['now_bal']:.2f}`\n" f"📊 Units: `{s['units']:+d}` | กำไรสุทธิ: `{profit:+.2f}`\n" f"━━━━━━━━━━━━━━━━━━━━\n" f"🛣 **เค้าไพ่ล่าสุด:** `{recent_road}`\n" f"📝 **ผลไม้ (W/L):** `{recent_wl}`\n\n" f"⚙️ *ตั้งค่าทุนพิมพ์ `/set 1000 10` (ทุน 1000 ไม้แรก 10)*" ) async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): chat_id = update.effective_chat.id user_sessions[chat_id] = init_session() await update.message.reply_text( render_message(chat_id), reply_markup=get_keyboard(), parse_mode="Markdown" ) async def set_money(update: Update, context: ContextTypes.DEFAULT_TYPE): chat_id = update.effective_chat.id try: start_bal = float(context.args) base_bet = float(context.args) user_sessions[chat_id] = init_session(start_bal, base_bet) await update.message.reply_text( render_message(chat_id), reply_markup=get_keyboard(), parse_mode="Markdown" ) except (IndexError, ValueError): await update.message.reply_text( "❌ วิธีตั้งค่า: พิมพ์ `/set [ทุนเริ่มต้น] [เงินไม้แรก]`\nตัวอย่าง: `/set 500 5`", parse_mode="Markdown" ) async def button_callback(update: Update, context: ContextTypes.DEFAULT_TYPE): query = update.callback_query await query.answer() chat_id = query.message.chat_id action = query.data if chat_id not in user_sessions: user_sessions[chat_id] = init_session() s = user_sessions[chat_id] if action == "RESET": user_sessions[chat_id] = init_session(s["start_bal"], s["base"]) elif action == "UNDO": if s["history"]: old_hist = s["history"][:-1] user_sessions[chat_id] = init_session(s["start_bal"], s["base"]) for move in old_hist: process_move(user_sessions[chat_id], move) elif action in ["P", "B", "T"]: process_move(s, action) await query.edit_message_text( text=render_message(chat_id), reply_markup=get_keyboard(), parse_mode="Markdown" ) def main(): if TOKEN == "YOUR_TELEGRAM_BOT_TOKEN": print("⚠️ คำเตือน: กรุณาใส่ TELEGRAM_BOT_TOKEN ก่อนรันบอท") app = Application.builder().token(TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(CommandHandler("set", set_money)) app.add_handler(CallbackQueryHandler(button_callback)) print("✅ บอทพร้อมทำงานแล้ว...") app.run_polling() if __name__ == "__main__": main()
